결과 폴링 처리

확장 작업은 10초 간격으로 결과를 폴링합니다. 진행 중 상태를 콜백으로 갱신하고, SUCCESS/COMPLETED 또는 ERROR/FAILED 상태로 종료를 판정하며, AbortSignal로 폴링을 취소할 수도 있습니다.

기능 개요

이미지 확장은 서버에서 오래 걸리는 작업이라 즉시 결과가 오지 않습니다. ResultArea.vue는 진입 시 확장 상태가 진행 중(status 0/1)이면 startPolling()으로 apiUtils.js의 pollingImageExtensionResultsWithCallback()을 호출합니다. 이 함수는 POLL_INTERVAL = 10000(10초) 간격으로 getImageExpansionResult(projectId, jobId)를 호출하며, 매 폴마다 전달받은 onUpdate 콜백에 전체 응답 데이터를 넘겨 진행률·부분 결과를 화면에 반영합니다.

종료 판정은 응답의 status로 합니다. SUCCESS 또는 COMPLETED면 resolve로 완료, ERROR 또는 FAILED면 reject로 실패 처리합니다. 폴링 취소는 AbortSignal로 지원하는데, ResultArea가 들고 있는 AbortController를 abort()하면 signal의 abort 리스너가 타이머를 정리하고 AbortError로 reject합니다. 컴포넌트는 이 에러를 정상 취소로 간주해 조용히 무시합니다.

두 가지 폴링 루프
이미지 확장은 apiUtils.js의 setTimeout 재귀 폴링(콜백·AbortSignal 지원)을 쓰고, 텍스트 제거는 PosterEdit.vue의 setInterval 10초 폴링을 따로 씁니다. 둘 다 10초 간격이지만 구현과 종료 조건(전자는 status 문자열, 후자는 result.imageUrl 존재)이 다릅니다.

흐름. 콜백으로 진행 상황을 흘려보내며 10초마다 재폴링하고, 종료 status에서 resolve/reject합니다. AbortController로 언제든 중단할 수 있습니다.

관계. ResultArea가 AbortController를 소유하고 폴링 함수에 signal을 넘깁니다. 폴링 함수는 getImageExpansionResult를 반복 호출합니다.

핵심 코드

posteredit/assets/js/utils/apiUtils.js — 10초 폴링 + 종료 판정 + AbortSignal
const POLL_INTERVAL = 10000; // 10 seconds

if (signal) {
    signal.addEventListener('abort', () => {
        clearTimeout(timeoutId);
        reject(new DOMException('Polling aborted', 'AbortError'));
    });
}

const poll = async () => {
    if (signal?.aborted) return;
    const response = await api.poster.getImageExpansionResult(projectId, jobId);
    const responseData = response.data;
    const { status } = responseData;

    if (onUpdate) onUpdate(responseData);                  // 진행률/부분결과 콜백

    if (status === 'ERROR' || status === 'FAILED') { clearTimeout(timeoutId); reject(new Error('Extension generation failed')); return; }
    if (status === 'SUCCESS' || status === 'COMPLETED') { clearTimeout(timeoutId); resolve(responseData); return; }

    timeoutId = setTimeout(poll, POLL_INTERVAL);           // 진행 중 → 재폴링
};
poll();
posteredit/components/ResultArea.vue — startPolling / stopPolling(AbortController)
async function startPolling() {
    if (isPolling.value) return
    isPolling.value = true
    abortController = new AbortController()

    await pollingImageExtensionResultsWithCallback(
        api, props.projectId, extensionId.value,
        (responseData) => {
            emit('extension-data-update', responseData)
            const { status, items } = responseData
            if (status === 'done')       { rawItems.value = items || []; stopPolling(); }
            else if (status === 'error') { rawItems.value = items || []; stopPolling(); }
            else { rawItems.value = [...(items || []), LOADING_CARD]; }
        },
        abortController.signal
    )
}

function stopPolling() {
    if (abortController) { abortController.abort(); abortController = null }
    isPolling.value = false
}
